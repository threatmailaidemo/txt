
import numpy as np
import os
import matplotlib.pyplot as plt

from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Step 1: Create Folder
os.makedirs("figs", exist_ok=True)

# Step 2: Load Dataset
data = load_iris()
X = data.data

print("Dataset shape:", X.shape)

# Step 3: Standardize Dataset
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 4: Calculate WCSS using Elbow Method
K_range = range(1, 11)
wcss = []

for k in K_range:
    km = KMeans(
        n_clusters=k,
        init="k-means++",
        n_init=10,
        random_state=42
    )

    km.fit(X_scaled)
    wcss.append(km.inertia_)

# Step 5: Display WCSS Results
print(f"\n{'K':<4}{'WCSS':>12}")

for k, w in zip(K_range, wcss):
    print(f"{k:<4}{w:>12.2f}")

# Step 6: Find Optimal K
d2 = np.diff(wcss, 2)
elbow_k = list(K_range)[int(np.argmax(d2)) + 1]

print("\nOptimal K using Elbow Method =", elbow_k)

# Step 7: Display Elbow Graph
plt.figure(figsize=(7, 5))

plt.plot(list(K_range), wcss, marker="o")

plt.axvline(
    elbow_k,
    linestyle="--",
    label=f"Elbow K = {elbow_k}"
)

plt.xlabel("Number of Clusters (K)")
plt.ylabel("WCSS")
plt.title("Elbow Method")
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.savefig("figs/exp8_elbow.png", dpi=150)
plt.show()

# Step 8: Create Final K-Means Model
final_k = 3

km = KMeans(
    n_clusters=final_k,
    init="k-means++",
    n_init=10,
    random_state=42
)

km.fit(X_scaled)

print(f"\nFinal model: K = {final_k}")
print("Cluster sizes:", np.bincount(km.labels_))
print("WCSS (inertia):", round(km.inertia_, 2))

# Step 9: Display Final Cluster Graph
plt.figure(figsize=(7, 5))

# Plot clusters using first two features
plt.scatter(
    X_scaled[:, 0],
    X_scaled[:, 1],
    c=km.labels_,
    cmap="viridis",
    s=40,
    edgecolor="black"
)

# Plot centroids
plt.scatter(
    km.cluster_centers_[:, 0],
    km.cluster_centers_[:, 1],
    marker="X",
    s=250,
    label="Centroids",
    edgecolor="black"
)

plt.xlabel("Sepal Length (Standardized)")
plt.ylabel("Sepal Width (Standardized)")
plt.title(f"K-Means Clusters (K = {final_k})")
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.savefig("figs/exp8_clusters.png", dpi=150)
plt.show()
